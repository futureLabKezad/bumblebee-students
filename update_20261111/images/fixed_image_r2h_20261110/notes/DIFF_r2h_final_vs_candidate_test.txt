+class B2Hold(unittest.TestCase):
+    """review of the final (S2): the medoid is published for _REENTRY_HOLD = 0.5 s after the first sample, then the
+    latest pose. After the pause every 3rd pose is +0.03 (the medoid of the last 3 is then 0.0, not the latest).
+    Control: the same scenario with a 0.05 s hold fails the bound."""
+
+    @staticmethod
+    def last_differing(mod):
+        frames = []
+
+        def det(t):
+            if t < 2.1:
+                return 2, (0.0, 0.0, 1.0)
+            if t < 3.1:
+                return 0, None
+            x = 0.03 if len(frames) % 3 == 2 else 0.0
+            frames.append((t, x))
+            return 2, (x, 0.0, 1.0)
+        s = Sim(mod, det)
+        s.run(5.0)
+        after = s.sent(after=frames[0][0])
+        t1 = after[0][0]
+        latest = lambda t: [x for tf, x in frames if tf < t][-1]       # timers fire before a frame at the same time
+        differ = [t for t, m in after if abs(m.pose.pose.position.x - latest(t)) > 1e-9]
+        return t1, differ
+
+    def test_hold_is_half_a_second(self):
+        t1, differ = self.last_differing(R2H)
+        self.assertTrue(differ)
+        self.assertGreaterEqual(max(differ), t1 + 0.3)
+        self.assertLess(max(differ), t1 + R2H._REENTRY_HOLD + 1e-6)
+        short = B.load(str(R2H_VPE), 'vpe_fix_r2h_hold_control')
+        short._REENTRY_HOLD = 0.05                                       # control only (module copy)
+        ct1, cdiffer = self.last_differing(short)
+        self.assertTrue(not cdiffer or max(cdiffer) < ct1 + 0.1, (ct1, cdiffer))

+class B2TwoPoses(unittest.TestCase):
+    """T44 (SHOULD of the candidate review): a slow detector (6 Hz) ends the re-entry wait with 2 fresh poses; the
+    medoid of 2 is a tie and must go to the NEWER pose. Control: the candidate sends the older one."""

+    @staticmethod
+    def det():
+        k = {'n': -1}

+        def d(t):
+            if 3.0 <= t < 4.5:
+                return 0, None
+            if t >= 4.5:
+                k['n'] += 1
+                return 2, (0.02 if k['n'] >= 1 else 0.0, 0.0, 1.0)     # back: x=0, then x=0.02

+    def first_after_pause(self, mod):
