+// T25 (r2d, 09.11.2026; r2e after the independent review): script lease watchdog decision for simple_offboard.
+// No ROS: unit-tested by test/test_script_lease.cpp of the r2e folder with plain g++.
+//
+// drone.py sends a lease every 0.5 s from its first flight command. The watchdog lands (simple_offboard's own
+// /land) when the lease is older than timeout while the vehicle is armed in OFFBOARD and the lease was seen fresh
+// while armed in OFFBOARD in this arm cycle, or an auto_arm request came with a fresh lease. Pilot wins: armed in
+// any other mode (pilot switch, PX4 failsafe, our own AUTO.LAND) disengages it until the next disarm.
+// r2e: it acts only on a State received after the lease became overdue, and only after it has ticked regularly
+// for 1 s: after a stall of this node the cached "armed in OFFBOARD" may be old (the reviewer's probe: a 1.8-2 s
+// SIGSTOP made it land over PX4's offboard-loss failsafe), and queued lease and State messages must drain first.
+#pragma once
+#include <cstdint>
+
+constexpr int64_t kScriptLeaseRetryNs = 1000000000LL;      // a failed /land is retried after 1 s
+constexpr int64_t kScriptLeaseTickGapNs = 500000000LL;     // a longer gap between ticks: this node stalled
+constexpr int64_t kScriptLeaseResumeNs = 1000000000LL;     // regular ticks needed before acting
+constexpr int64_t kScriptLeaseArmWindowNs = 15000000000LL; // an auto_arm request engages an arm within this
+
+struct ScriptLease {
+    bool engaged = false;          // the lease belongs to this arm cycle
+    bool stepped_back = false;     // armed outside OFFBOARD after engaging: off until disarmed
+    bool armed_seen = false;       // this arm cycle has had a tick
+    int64_t last_try_ns = 0;       // last landing attempt, 0 = none
+    int64_t last_tick_ns = 0;      // previous tick, 0 = none
+    int64_t regular_since_ns = 0;  // ticks no more than kScriptLeaseTickGapNs apart since then
+};
+
+// One watchdog tick. state_ok: FCU state connected and fresh (an unknown state never acts).
+// lease_rx_ns: receipt of the last lease; state_rx_ns: receipt of the State that armed/offboard come from;
+// arm_req_ns: last auto_arm request made while the lease was fresh. All steady clock, 0 = never. True = land now.
+inline bool scriptLeaseLandDue(ScriptLease& l, bool state_ok, bool armed, bool offboard,
+                               int64_t lease_rx_ns, int64_t state_rx_ns, int64_t arm_req_ns,
+                               int64_t now_ns, int64_t timeout_ns, int64_t retry_ns) {
+    if (l.last_tick_ns == 0 || now_ns - l.last_tick_ns > kScriptLeaseTickGapNs) l.regular_since_ns = now_ns;
+    l.last_tick_ns = now_ns;
+    if (timeout_ns <= 0 || !state_ok) return false;
+    if (!armed) {
+        l.engaged = l.stepped_back = l.armed_seen = false;
+        l.last_try_ns = 0;
+        return false;
+    }
+    const bool first_armed_tick = !l.armed_seen;
+    l.armed_seen = true;
+    if (!offboard) {
+        if (l.engaged) l.stepped_back = true;
+        l.engaged = false;
+        return false;
+    }
+    if (l.stepped_back) return false;
+    if (first_armed_tick && arm_req_ns != 0 && now_ns - arm_req_ns <= kScriptLeaseArmWindowNs) l.engaged = true;
+    // A receipt after now (another thread) is fresh too
+    const bool fresh = lease_rx_ns != 0 && now_ns - lease_rx_ns <= timeout_ns;
+    if (fresh) { l.engaged = true; return false; }
+    if (!l.engaged) return false;
+    if (now_ns - l.regular_since_ns < kScriptLeaseResumeNs) return false;
+    const int64_t base = lease_rx_ns > arm_req_ns ? lease_rx_ns : arm_req_ns;
+    if (state_rx_ns <= base + timeout_ns || state_rx_ns < l.regular_since_ns) return false;
+    if (l.last_try_ns != 0 && now_ns - l.last_try_ns < retry_ns) return false;
+    l.last_try_ns = now_ns;
+    return true;
+}

+"""
+vpe_fix.py -- VPE publisher AND the sole PX4-parameter gate.
+
+ROLES
+=====
+1. Publishes /mavros/mavros/pose_cov at 30 Hz from the latest ArUco map pose
+   (with quality gating, soft fallback, and sanity guards).
+2. Exposes /vpe_fix/set_param (bumblebee/srv/SetParam) -- the ONLY channel
+   that writes PX4 parameters in this stack.
+
+ARCHITECTURAL RULE
+==================
+No other process writes PX4 params. set_params.py and preflight.py are
+read-only diagnostics. drone.set_param() in flight scripts publishes to
+/vpe_fix/set_param (not /mavros/mavros/set). PX4 config set manually
+through QGC / mavlink_shell is sacred -- the gate never clobbers it at
+boot, and restores it on shutdown for any param it touched.
+
+GATE BEHAVIOUR (rewritten 2026-06-05)
+=====================================
+The gate is honest, per-parameter. Original values are read from a cache
+populated by /parameter_events (MAVROS 2.x exposes PX4 params as ROS2
+parameters of node /mavros/mavros).
+
+  - At startup: trigger ParamPull(force_pull=true), subscribe to
+    /parameter_events, wait for cache to fill (>=800 params or 30 s).
+  - On set_param:
+      - if cache is not warm yet  -> reject with "cache warming up";
+      - if param not in cache     -> reject "param '<n>' not in PX4 cache";
+      - if cache value == request -> noop (no PX4 write, no journal);
+      - else                      -> record (param_id -> original) in journal
+                                     ONCE (first-write wins), then ParamSetV2.
+  - Revert paths are point-precise (no MAV_CMD_PREFLIGHT_STORAGE reload):
+      - on ARMED->DISARMED edge: ParamSetV2(original) for each non-persistent
+        override; persistent overrides are left in place. The edge is taken
+        only between States of a connected FCU (FCU STATE below); a
+        disconnected State is never a disarm;
+      - on TTL fire: ParamSetV2(original) for that one param;
+      - on shutdown / explicit /vpe_fix/revert_all: every override.
+
+Why this matters: the previous implementation used MAV_CMD_PREFLIGHT_STORAGE
+param1=0 to reload PX4 RAM from EEPROM on every disarm. That assumed EEPROM
+held the operator's pristine config -- but EEPROM can be polluted (PX4
+firmware autosave behaviour, QGC "Save" clicks, old baseline writers). The
+result was profile values leaking from auto-flight into subsequent manual
+flights.
+
+SAFETY
+======
+The gate refuses EKF2_EV_CTRL=0 while armed. Toggling EV_CTRL=0
+mid-flight caused IMU-only dead-reckoning runaway on 2026-05-13.
+
+UNKNOWN WRITE OUTCOME (H2, 2026-10-30, round 2)
+===============================================
+A failed ParamSetV2 call does not say whether PX4 took the value: future
+timeout, MAVROS gone or respawned mid-call, an exception, or success=False
+after MAVROS's own resends (bench session 1: MAVROS died on a param write,
+PX4 had applied it, the gate dropped the original and nothing restored it).
+Only "service not available" before every call means no request was sent,
+so PX4 surely did not change: then the first-write journal entry is dropped
+as before. After any other failure of a first write the entry is KEPT and
+marked unconfirmed:
+  - nothing but a write of its original removes it: the disarm edge,
+    revert_all, TTL or shutdown, as for any override (writing the original
+    to a PX4 that already holds it changes nothing). No cache value and no
+    /parameter_events update drops it, whatever PX4 seems to show;
+  - it is never written while armed: no TTL is armed for it in the air, and
+    a TTL, revert_all or shutdown that comes while armed leaves it for the
+    disarm edge. So the change adds no in-air ParamSet to 7f107882;
+  - a later noop request that finds it logs it again (information only);
+  - a later successful write of the same param makes it an ordinary
+    override (first-write original kept).
+
+VPE PUBLISH LOGIC
+=================
+Vision flow is gated by whether vpe_fix publishes /mavros/mavros/pose_cov.
+EKF2_EV_CTRL is not toggled in flight -- when the publish flag is False,
+the 30 Hz publish_timer emits nothing and PX4 EKF coasts on baro+IMU until
+fresh vision returns.
+
+  Publish = True when the FCU state is known (below) and:
+    - Drone is DISARMED (pre-arm warm-up; quality ignored on the ground).
+    - ARMED + OFFBOARD + (multi_marker_fresh OR single_marker_fresh).
+
+  Publish = False when:
+    - FCU state unknown, whatever the last known armed/mode.
+    - ARMED in non-OFFBOARD mode.
+    - ARMED + OFFBOARD + vision bad. Baro is secondary altitude source.
+
+FCU STATE (P1/P1b fix, 2026-10-21)
+==================================
+armed/mode come only from a /mavros/state with connected=True whose header
+stamp is current (not zero, at most _STATE_STAMP_MAX_AGE old: an older one
+is a latched replay and is ignored). The FCU state is known while the last
+such State was received at most _STATE_STALE_TIMEOUT ago; otherwise the
+gate is closed:
+  - connected=False is MAVROS's placeholder (armed=False, mode='') at its
+    own start and on its conn_timeout, not PX4 data: it closes the gate at
+    once, keeps the last reported armed/mode, and is never a disarm (no
+    parameter revert in the air);
+  - no State for _STATE_STALE_TIMEOUT (MAVROS died or the FCU link went
+    silent): the gate is closed within one publish tick, so a respawned
+    MAVROS never passes the stream on for the old state.
+The gate is re-evaluated at once on every change of armed or mode and on
+the first State after an unknown period, not only on armed edges.
+
+OPERATOR NOTE
+=============
+EKF2_EV_CTRL must be set on the FCU itself (via QGC) to 11 (hpos+vpos+yaw)
+or 15 (hpos+vpos+vel+yaw). If a flight script needs a guaranteed value,
+call drone.set_param('EKF2_EV_CTRL', 11, persistent=True) once at start:
+the gate reads PX4, writes only if it differs, and leaves it set.

+    # ==================================================================
+    # PX4 param cache (fed by /parameter_events)
+    # ==================================================================
+
+    def _on_parameter_event(self, msg):
+        # /parameter_events carries events from every node in the graph.
+        # Filter for MAVROS's plugin container, which owns PX4 params.
+        if msg.node != _MAVROS_NODE:
+            return
+        with self._cache_lock:
+            for p in msg.new_parameters:
+                v = self._pv_to_cache(p.value)
+                if v is not None:
+                    self._px4_cache[p.name] = v
+            for p in msg.changed_parameters:
+                v = self._pv_to_cache(p.value)
+                if v is not None:
+                    self._px4_cache[p.name] = v
+            # NOTE: deleted_parameters intentionally ignored. MAVROS 2.x
+            # publishes delete+add cycles around its own set_parameter calls
+            # (and during ParamPull), which would race with our writes and
+            # transiently empty the cache. PX4 params are never *really*
+            # deleted in the operational sense; we only need to track latest
+            # known values, so a stale value is preferable to "not in cache".
+            n = len(self._px4_cache)
+            if not self._cache_hot and n >= _CACHE_HOT_MIN:
+                self._cache_hot = True
+                self.get_logger().info(
+                    'vpe_fix gate: cache HOT (%d params)' % n)
+
+    @staticmethod
+    def _pv_to_cache(pv):
+        """Convert rcl_interfaces ParameterValue to (rcl_type, value)."""
+        t = int(pv.type)
+        if t == _RCL_PT_INTEGER:
+            return (_RCL_PT_INTEGER, int(pv.integer_value))
+        if t == _RCL_PT_DOUBLE:
+            return (_RCL_PT_DOUBLE, float(pv.double_value))
+        # PX4 params are int or double; ignore other types.
+        return None
+
+    def _initial_pull_tick(self):
+        if self._initial_pull_done:
+            return
+        self._initial_pull_done = True
+        if not self.param_pull_client.wait_for_service(timeout_sec=2.0):
+            self.get_logger().warn(
+                'vpe_fix gate: ParamPull service unavailable, '
+                'cache will fill from passive events only')
+            return
+        self.get_logger().info('vpe_fix gate: triggering initial ParamPull')
+        ok = self._param_pull_blocking(timeout=10.0)
+        with self._cache_lock:
+            n = len(self._px4_cache)
+        self.get_logger().info(
+            'vpe_fix gate: initial ParamPull %s, cache=%d'
+            % ('OK' if ok else 'FAILED', n))
+
+    def _warmup_timeout_tick(self):
+        if self._warmup_timeout_done:
+            return
+        self._warmup_timeout_done = True
+        with self._cache_lock:
+            n = len(self._px4_cache)
+            if self._cache_hot:
+                return
+            if n >= _CACHE_HOT_MIN_TIMEOUT:
+                self._cache_hot = True
+                self.get_logger().warn(
+                    'vpe_fix gate: cache HOT after timeout '
+                    '(%d params, expected >=%d)' % (n, _CACHE_HOT_MIN))
+            else:
+                # Mark hot anyway; per-name lookups will fail with a useful
+                # message instead of hanging forever.
+                self._cache_hot = True
+                self.get_logger().error(
+                    'vpe_fix gate: cache warm-up degraded (%d params, '
+                    'expected >=%d); per-param lookups may fail' % (n, _CACHE_HOT_MIN_TIMEOUT))
+
+    def _update_cache_from_write(self, name, value, ptype):
+        """After our own successful ParamSetV2, push the new value into the
+        cache immediately so subsequent reads don't depend on /parameter_events
+        round-trip (which sometimes drops or delays under MAVROS load)."""
+        rcl_type = _RCL_PT_INTEGER if ptype == _PTYPE_INT else _RCL_PT_DOUBLE
+        stored = int(round(value)) if ptype == _PTYPE_INT else float(value)
+        with self._cache_lock:
+            self._px4_cache[name] = (rcl_type, stored)
+
+    # ==================================================================
+    # PX4-param gate
+    # ==================================================================
+
+    def _handle_set_param(self, request, response):
+        name = request.param_id
+        value = float(request.value)
+        ptype = int(request.ptype)
+        ttl = float(request.ttl_sec)
+        persistent = bool(request.persistent)
+
+        # --- safety: never let EKF2_EV_CTRL=0 land in PX4 while armed ---
+        if name == 'EKF2_EV_CTRL' and int(round(value)) == 0 and self.armed:
+            response.success = False
+            response.message = ('refused: EKF2_EV_CTRL=0 while armed is unsafe '
+                                '(2026-05-13 incident)')
+            response.original_value = float('nan')
+            self.get_logger().warn('vpe_fix gate: %s' % response.message)
+            return response
+
+        if not self._cache_hot:
+            response.success = False
+            response.message = 'cache warming up'
+            response.original_value = float('nan')
+            return response
+
+        with self._cache_lock:
+            cache_entry = self._px4_cache.get(name)
+        if cache_entry is None:
+            response.success = False
+            response.message = "param '%s' not in PX4 cache" % name
+            response.original_value = float('nan')
+            self.get_logger().warn('vpe_fix gate: %s' % response.message)
+            return response
+
+        _cur_type, cur_value = cache_entry
+
+        # Skip-if-equal: PX4 already holds the requested value. No write, no
+        # journal entry, no revert on disarm.
+        if self._values_equal(cur_value, value, ptype):
+            response.success = True
+            response.message = 'noop: already %g' % cur_value
+            response.original_value = float(cur_value)
+            # H2: an override with an unknown write outcome stays in the
+            # journal whatever the cache shows (UNKNOWN WRITE OUTCOME): log.
+            with self._overrides_lock:
+                kept = self._overrides.get(name)
+                if kept is not None and not kept.unconfirmed:
+                    kept = None
+            if kept is not None:
+                self.get_logger().info(
+                    'vpe_fix gate: noop %s=%g, override with unknown write '
+                    'outcome kept UNCONFIRMED (orig=%g, requested %g): '
+                    'restored on disarm, revert_all or TTL'
+                    % (name, value, kept.original, kept.requested))
+            return response
+
+        # First-write wins: only record original on the FIRST override of a
+        # given param. Subsequent writes (e.g. profile A then profile B) keep
+        # the same "golden" original so disarm restores all the way back.
+        is_first_write = False
+        with self._overrides_lock:
+            entry = self._overrides.get(name)
+            if entry is None:
+                entry = _Override(
+                    original=float(cur_value), ptype=ptype,
+                    persistent=persistent, ttl_timer=None)
+                self._overrides[name] = entry
+                is_first_write = True
+            else:
+                entry.persistent = persistent
+                if entry.ttl_timer is not None:
+                    entry.ttl_timer.cancel()
+                    entry.ttl_timer = None
+            original = entry.original
+
+        ok, msg = self._param_set_with_retry(name, value, ptype)
+        if not ok:
+            if is_first_write and msg == _SET_NOT_SENT:
+                # No request was sent: PX4 surely unchanged, drop the entry.
+                with self._overrides_lock:
+                    self._overrides.pop(name, None)
+                self.get_logger().warn(
+                    'vpe_fix gate: set %s=%g failed, not sent (%s); PX4 '
+                    'unchanged, no override kept' % (name, value, msg))
+            elif is_first_write:
+                # H2: outcome unknown, PX4 may hold the value. Keep the
+                # original, unconfirmed, until a write of it (UNKNOWN WRITE
+                # OUTCOME); put it back if a revert took it during the call.
+                with self._overrides_lock:
+                    entry.unconfirmed = True
+                    entry.requested = value
+                    self._overrides.setdefault(name, entry)
+                self.get_logger().warn(
+                    'vpe_fix gate: set %s=%g failed, outcome unknown (%s); '
+                    'override kept UNCONFIRMED (orig=%g): restored on disarm, '
+                    'revert_all or TTL, never while armed'
+                    % (name, value, msg, original))
+                # No TTL armed in the air (no in-air write beyond 7f107882).
+                if ttl > 0 and not persistent and not self.armed:
+                    self._arm_ttl(name, ttl)
+            response.success = False
+            response.message = msg
+            response.original_value = float('nan')
+            return response
+
+        self._update_cache_from_write(name, value, ptype)
+        with self._overrides_lock:
+            if name in self._overrides:
+                # H2: a successful write is a known outcome.
+                self._overrides[name].unconfirmed = False
+
+        if ttl > 0 and not persistent:
+            self._arm_ttl(name, ttl)
+
+        response.success = True
+        response.message = ('applied %s=%g%s'
+                            % (name, value, ', persistent' if persistent else ''))
+        response.original_value = float(original)
+        self.get_logger().info(
+            'vpe_fix gate: %s (orig=%g)' % (response.message, original))
+        return response
+
+    @staticmethod
+    def _values_equal(a, b, ptype):
+        if ptype == _PTYPE_INT:
+            return int(round(a)) == int(round(b))
+        return abs(float(a) - float(b)) < 1e-6
+
+    def _handle_revert_all(self, request, response):
+        with self._overrides_lock:
+            count = len(self._overrides)
+        if count == 0:
+            response.success = True
+            response.message = 'no overrides to revert'
+            return response
+        self._revert_all(reason='explicit_revert_all')
+        response.success = True
+        response.message = ('reverted %d override(s) point-precise' % count)
+        return response
+
+    def _param_set_with_retry(self, name, value, ptype):
+        """Issue ParamSetV2 once; if it fails, retry once after a short pause.
+
+        We deliberately do NOT force-pull between attempts: ParamPull floods
+        /parameter_events for several seconds, which saturates the executor
+        and causes the second ParamSetV2 future to never complete (its done
+        callback can't get a thread). The thinner retry path is more reliable.
+        """
+        ok, msg = self._param_set_once(name, value, ptype, timeout=_PARAM_SET_TIMEOUT)
+        if ok:
+            return True, 'ok'
+        self.get_logger().warn(
+            'vpe_fix gate: set %s=%g failed (attempt 1, %s); retrying in 0.3 s'
+            % (name, value, msg))
+        # Short pause lets MAVROS's queue settle.
+        threading.Event().wait(0.3)
+        ok, msg2 = self._param_set_once(name, value, ptype, timeout=_PARAM_SET_TIMEOUT)
+        if ok:
+            return True, 'ok on retry'

+            # H2: attempt 1 may have landed; keep that in the result.
+            msg2 = '%s (attempt 1: %s)' % (msg2, msg)
+        return False, msg2

+    def _param_set_once(self, name, value, ptype, timeout=_PARAM_SET_TIMEOUT):
+        if not self.param_set_client.wait_for_service(timeout_sec=timeout):
+            return False, 'service /mavros/mavros/set not available'
+        pval = ParameterValue()
+        if ptype == _PTYPE_INT:
+            pval.type = ParameterType.PARAMETER_INTEGER
+            pval.integer_value = int(round(value))
+        else:
+            pval.type = ParameterType.PARAMETER_DOUBLE
+            pval.double_value = float(value)
+        req = ParamSetV2.Request()
+        req.param_id = name
+        req.value = pval
+        fut = self.param_set_client.call_async(req)
+        evt = threading.Event()
+        fut.add_done_callback(lambda _f: evt.set())
+        if not evt.wait(timeout=timeout):
+            # Future never resolved -- usually means MAVROS is busy or the
+            # client callback group lost its thread to /parameter_events.
+            return False, 'future timeout after %.1fs' % timeout
+        try:
+            res = fut.result()
+        except Exception as e:

+        if res is None:
+            return False, 'future returned None'
+        if not res.success:

+        fut.add_done_callback(lambda _f: evt.set())

+    def _arm_ttl(self, name, ttl_sec):

+        with self._overrides_lock:

+            entry.ttl_timer = self.create_timer(ttl_sec, _fire)

+    def _restore_one(self, name, reason='manual'):
+        """Restore a single param to its captured original via ParamSetV2."""
+        with self._overrides_lock:
+            entry = self._overrides.get(name)
+            # H2: an unconfirmed original is never written while armed (no
+            # in-air write beyond 7f107882): kept for the disarm edge.
+            held = entry is not None and entry.unconfirmed and self.armed
+            if held:

+            else:

+            return

+                try:
+                    timer.cancel()
+                except Exception:
+                    pass

+                'vpe_fix gate: restore of %s=%g (%s) held: write outcome '
+                'unknown and FCU armed, restored on disarm'

+            try:

+            except Exception:

+    def _revert_all_non_persistent(self, reason):
+        with self._overrides_lock:

+            % (len(to_revert), reason, to_revert))

+            return

+            % (len(names), reason, names))

+            return False

+    def _state_clock_ns(self):

+        monotonicNs(): the simulation clock under use_sim_time, else the
+        steady clock (immune to NTP steps on the drone). None while the

+    def _fcu_state_unknown(self):

+            if self._state_rx_ns is None:
+                return 'no connected /mavros/state yet'
+            return 'MAVROS reports the FCU disconnected'

+        age = (now - self._state_rx_ns) / 1e9
+        if not 0.0 <= age <= _STATE_STALE_TIMEOUT:
+            # a clock that went backwards is no evidence either
+            return 'no /mavros/state for %.2fs' % age
+        return None

+    def _evaluate_gate(self):
+        unknown = self._fcu_state_unknown()
+        if unknown is not None:
+            return False, 'FCU state unknown: %s (no publish)' % unknown
+        if not self.armed:
+            return True, 'disarmed (warm-up)'
+        if not self.in_offboard:
+            return False, 'armed in manual (no publish)'
+        if self._multi_marker_fresh() or self._single_marker_fresh():
+            self._last_vision_ok_time = self.get_clock().now()
+            return True, 'OFFBOARD + vision_ok'
+        if self._last_vision_ok_time is not None:
+            age = (self.get_clock().now() - self._last_vision_ok_time).nanoseconds / 1e9
+            if age < _GATE_HYSTERESIS:
+                return True, 'OFFBOARD + recent vision ok (coasting %.1fs)' % age
+        return False, 'OFFBOARD + vision_bad (publish suspended)'

+    def _apply_gate(self, force=False):
+        publish, reason = self._evaluate_gate()
+        if publish != self.vpe_active or force:
+            self.vpe_active = publish
+            self.get_logger().info(
+                'vpe_fix: %s -> publish=%s' % (reason, publish))

+    def state_callback(self, msg):
+        was_fresh = self._fcu_state_unknown() is None
+        was_connected = self.fcu_connected
+        if not msg.connected:
+            # P1: MAVROS sys_status publishes connected=False, armed=False,
+            # mode='' when it (re)starts and on its conn_timeout. These are
+            # placeholders, not PX4 data: keep the last reported armed/mode
+            # (never a disarm, so no parameter revert in the air) and close
+            # the gate at once.

+                    'vpe_fix: /mavros/state connected=False (MAVROS start or '
+                    'FCU link timeout): FCU state unknown, keeping last '
+                    'reported armed=%s offboard=%s, no parameter revert'
+                    % (self.armed, self.in_offboard))
+                self._apply_gate()
+            return

+        stamp_age = (self._state_clock.now().nanoseconds - stamp_ns) / 1e9

+            # An old latched replay (or a State before /clock in SITL) is no
+            # report from the FCU (simple_offboard handleState): ignore it;
+            # the last State taken keeps deciding until it is stale.
+            self._log_state_ignored(stamp_ns)

+        was_offboard = self.in_offboard

+        self.in_offboard = (msg.mode == 'OFFBOARD')

+            self.get_logger().info(
+                'vpe_fix: FCU state known: armed=%s mode=%s'
+                % (self.armed, msg.mode))

+        # knowledge (c33: armed edges only; mode changes waited for the tick).
+        if (self.armed != was_armed or self.in_offboard != was_offboard
+                or not was_fresh):
+            self._apply_gate()

+                # ARMED -> DISARMED edge between States of a connected FCU:
+                # point-precise restore of every non-persistent override.
+                self._revert_all_non_persistent(reason='disarmed')
+
+    def _log_state_ignored(self, stamp_ns):
+        self._state_ignored += 1
+        now = time.monotonic_ns()
+        last = self._last_state_ignored_log
+        if last is None or now - last >= 2000000000:
+            self._last_state_ignored_log = now
+            self.get_logger().warn(
+                'vpe_fix: ignored connected /mavros/state stamped %.3f s (zero, '
+                'older than %.1fs or no clock yet; %d ignored)'
+                % (stamp_ns / 1e9, _STATE_STAMP_MAX_AGE, self._state_ignored))

+    def _maybe_reseed(self, msg_pose, now, reason):
+        """Soft-reject helper: count consecutive rejects and reseed last_pose
+        if the gate appears self-locked (z-jump self-lock incident 2026-05-26).
+        Always still drops the current frame; reseed only affects future
+        comparisons. Hard altitude guards (floor/ceiling) must NOT call this.
+
+        Reseed hardening (2026-06-10 drift fix): a single rejected pose is no
+        longer accepted wholesale -- one IPPE mirror-flip frame used to become
+        the 30 Hz republished baseline and snap the EKF. A reseed now requires
+        _RESEED_CONSISTENT_N recent rejected poses that mutually agree; the
+        medoid of them becomes the new baseline. The escape hatch after
+        _RESEED_FORCE_REJECTS consecutive rejects keeps the self-heal
+        guarantee: reseed always eventually happens."""
+        self._consec_rejects += 1
+        self._reseed_candidates.append((now, msg_pose))
+        count_limit = 7 if self.armed else 5
+        age_limit = 0.5 if self.armed else 1.0
+        if self.last_pose_time is not None:
+            age = (now - self.last_pose_time).nanoseconds / 1e9
+        else:
+            age = float('inf')
+        if not (self._consec_rejects >= count_limit or age > age_limit):
+            return
+        cands = [(t, p) for (t, p) in self._reseed_candidates
+                 if (now - t).nanoseconds / 1e9 <= _RESEED_WINDOW]
+        consistent = (len(cands) >= _RESEED_CONSISTENT_N
+                      and self._candidates_consistent(cands))
+        forced = self._consec_rejects >= _RESEED_FORCE_REJECTS
+        if not (consistent or forced):
+            now_s = now.nanoseconds * 1e-9
+            if now_s - self._last_reseed_defer_log >= 1.0:
+                self._last_reseed_defer_log = now_s
+                self.get_logger().warn(
+                    'vpe_fix: reseed deferred (%s, %d/%d consistent candidates, age=%.2fs)'
+                    % (reason, len(cands), _RESEED_CONSISTENT_N, age))
+            return
+        seed = self._medoid([p for _, p in cands]) if cands else msg_pose
+        self.last_pose = seed
+        self.last_pose_time = now
+        self.last_cov = None
+        self._consec_rejects = 0
+        self._reseed_candidates.clear()
+        self.get_logger().warn(
+            'vpe_fix: gate RESEEDED (%s%s) age=%.2fs new_xyz=(%.3f,%.3f,%.3f)'
+            % (reason, ' FORCED' if forced and not consistent else '', age,
+               seed.position.x, seed.position.y, seed.position.z))

+    @staticmethod
+    def _candidates_consistent(cands):
+        for i in range(len(cands)):
+            for j in range(i + 1, len(cands)):
+                p, q = cands[i][1], cands[j][1]
+                dxy = ((p.position.x - q.position.x) ** 2 +
+                       (p.position.y - q.position.y) ** 2) ** 0.5
+                if dxy > _RESEED_XY_CONSIST:
+                    return False
+                if abs(p.position.z - q.position.z) > _RESEED_Z_CONSIST:
+                    return False
+        return True
+
+    @staticmethod
+    def _medoid(poses):
+        """Pose minimizing the summed 3D distance to the other candidates."""
+        best, best_cost = None, None
+        for p in poses:
+            cost = 0.0
+            for q in poses:
+                cost += ((p.position.x - q.position.x) ** 2 +
+                         (p.position.y - q.position.y) ** 2 +
+                         (p.position.z - q.position.z) ** 2) ** 0.5
+            if best_cost is None or cost < best_cost:
+                best, best_cost = p, cost
+        return best

+    def aruco_map_callback(self, msg):

+        if z < _Z_FLOOR_GUARD or z > _Z_CEILING_GUARD:
+            self.get_logger().warn(

+                self._maybe_reseed(msg.pose.pose, now, 'z-jump')
+                return

+            dy = msg.pose.pose.position.y - self.last_pose.position.y
+            horiz = (dx * dx + dy * dy) ** 0.5

+                    'vpe_fix: rejected pose xy-jump %.3f (>%.2f)' % (horiz, _XY_JUMP_MAX))
+                self._maybe_reseed(msg.pose.pose, now, 'xy-jump')
+                return

+        self._consec_rejects = 0
+        self._reseed_candidates.clear()
+        if any(msg.pose.covariance):

+            cov[0]  = xy
+            cov[7]  = xy
+            cov[14] = z
+            cov[21] = 1e6
+            cov[28] = 1e6
+            cov[35] = _YAW_VARIANCE
+            out.pose.covariance = cov

+        self.vpe_pub.publish(out)
+        self.published += 1
+        if self.published % 90 == 1:
+            self.get_logger().info(
+                'vpe_fix: published %d | z=%.3f | markers=%s | soft=%s' % (
+                    self.published, out.pose.pose.position.z,
+                    self.visible_ids, soft))

+Content (review of the final, S1): the declared parts are also pinned to the accepted final r2h (sha256 of their
+ast.dump; ast.dump is stable within a Python minor version: 3.12 = ROS 2 Jazzy, PC and drone). A change of a declared
+part therefore needs a new pin round, like any other part.
+
+Env: PIN_TARGET (file under test), PIN_R2G (the r2g copy), PIN_HARNESS (tests_existing/harness, for rosstub and
+_EXPECTED_CONSTANTS, as in test_pins_r2.py)

+def load(path, name):
+    spec = importlib.util.spec_from_file_location(name, path)
+    mod = importlib.util.module_from_spec(spec)
+    spec.loader.exec_module(mod)
+    return mod
+
+
+def tree(path):
+    return ast.parse(open(path).read())
+
+
+def parts(t):
+    """every top-level part: constants, imports, classes, functions, VpeFix methods, VpeFix non-method body, other
+    statements"""
+    out = {}
+    for n in t.body:
+        if isinstance(n, ast.Assign):
+            for n2 in n.targets:
+                if isinstance(n2, ast.Name):
+                    out['const ' + n2.id] = ast.dump(n.value)
+        elif isinstance(n, (ast.Import, ast.ImportFrom)):
+            out['import ' + ast.dump(n)] = ''
+        elif isinstance(n, ast.ClassDef) and n.name == 'VpeFix':
+            out['class VpeFix bases'] = repr([ast.dump(b) for b in n.bases])
+            out['class VpeFix non-method body'] = repr([ast.dump(m) for m in n.body if not isinstance(m, ast.FunctionDef)])
+            for m in n.body:
+                if isinstance(m, ast.FunctionDef):
+                    out['VpeFix.' + m.name] = ast.dump(m)
+        elif isinstance(n, (ast.FunctionDef, ast.ClassDef)):
+            out[n.name] = ast.dump(n)
+        else:
+            out['stmt ' + ast.dump(n)] = ''
+    return out
+
+
+def method(path, name):
+    return [m for n in tree(path).body if isinstance(n, ast.ClassDef) and n.name == 'VpeFix' for m in n.body
+            if isinstance(m, ast.FunctionDef) and m.name == name][0]

+class PinsR3(unittest.TestCase):
+    def test_r3_reference_is_r2g(self):
+        with open(R2G, 'rb') as h:
+            self.assertTrue(hashlib.sha256(h.read()).hexdigest().startswith(R2G_SHA))
+
+    def test_r3_only_the_declared_parts_differ_from_r2g(self):
+        a, b = parts(tree(R2G)), parts(tree(TARGET))
+        self.assertEqual({k for k in a if k in b and a[k] != b[k]}, CHANGED)
+        self.assertEqual(set(b) - set(a), ADDED)
+        self.assertEqual(set(a) - set(b), set())
+
+    def test_r3_declared_parts_are_the_accepted_final(self):
+        self.assertEqual(set(DIGESTS), CHANGED | ADDED)
+        b = parts(tree(TARGET))
+        for k, v in DIGESTS.items():
+            self.assertEqual(hashlib.sha256(b[k].encode()).hexdigest(), v, k)
+
+    def test_r3_aruco_map_callback_one_added_statement(self):
+        sa = [ast.dump(s) for s in method(R2G, 'aruco_map_callback').body]
+        sb = [ast.dump(s) for s in method(TARGET, 'aruco_map_callback').body]
+        self.assertEqual(len(sb), len(sa) + 1)
+        extra = [s for s in sb if s not in sa]
+        self.assertEqual(len(extra), 1)
+        self.assertIn("attr='_recent_poses'", extra[0])
+        self.assertEqual([s for s in sb if s in sa], sa)
+
+    def test_r3_p1_gate_h2_and_guard_code_unchanged(self):
+        for m in ('_evaluate_gate', '_apply_gate', 'state_callback', '_fcu_state_unknown', '_state_clock_ns',
+                  '_state_watchdog_tick', '_gate_tick', '_multi_marker_fresh', '_medoid', '_handle_set_param',
+                  '_param_set_with_retry', '_restore_one', '_param_set_once'):
+            self.assertEqual(ast.dump(method(R2G, m)), ast.dump(method(TARGET, m)), m)
+
+    def test_r3_constants(self):
+        a, b = parts(tree(R2G)), parts(tree(TARGET))
+        for k, v in a.items():
+            if k.startswith('const '):
+                self.assertEqual(b.get(k), v, k)
+        mod = load(TARGET, 'pin_r3_target')
+        for k, v in REENTRY.items():
+            self.assertEqual(getattr(mod, k), v, k)
+        fresh = load(os.path.join(HARNESS, 'vpe_freshness_optin_c09.py'), 'pin_r3_fresh')
+        for k, v in fresh._EXPECTED_CONSTANTS.items():
+            self.assertEqual(getattr(mod, k), v, k)
